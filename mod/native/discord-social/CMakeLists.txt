cmake_minimum_required(VERSION 3.21)
project(NewYearDiscordSocial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DISCORD_SDK_ROOT "" CACHE PATH "Official Discord Social SDK folder containing include, lib and bin")
set(SDK_PLATFORM_SUFFIX "" CACHE STRING "SDK platform subdirectory, e.g. /arm64")
set(DISCORD_SDK_LIBRARY "" CACHE FILEPATH "Override SDK import/shared library")
set(DISCORD_SDK_RUNTIME "" CACHE FILEPATH "Override SDK runtime DLL/shared library")
if(NOT EXISTS "${DISCORD_SDK_ROOT}/include/discordpp.h")
    message(FATAL_ERROR "Download the C++ SDK from your application's Discord Social SDK > Downloads, and set DISCORD_SDK_ROOT.")
endif()
add_executable(newyear-discord-social main.cpp)
target_include_directories(newyear-discord-social PRIVATE "${DISCORD_SDK_ROOT}/include")
if(NOT DISCORD_SDK_LIBRARY)
    if(WIN32)
        set(DISCORD_SDK_LIBRARY "${DISCORD_SDK_ROOT}/lib/release${SDK_PLATFORM_SUFFIX}/discord_partner_sdk.lib")
    elseif(APPLE)
        set(DISCORD_SDK_LIBRARY "${DISCORD_SDK_ROOT}/lib/release${SDK_PLATFORM_SUFFIX}/libdiscord_partner_sdk.dylib")
    else()
        set(DISCORD_SDK_LIBRARY "${DISCORD_SDK_ROOT}/lib/release${SDK_PLATFORM_SUFFIX}/libdiscord_partner_sdk.so")
    endif()
endif()
if(NOT DISCORD_SDK_RUNTIME)
    if(WIN32)
        set(DISCORD_SDK_RUNTIME "${DISCORD_SDK_ROOT}/bin/release${SDK_PLATFORM_SUFFIX}/discord_partner_sdk.dll")
    else()
        set(DISCORD_SDK_RUNTIME "${DISCORD_SDK_LIBRARY}")
    endif()
endif()
target_link_libraries(newyear-discord-social PRIVATE "${DISCORD_SDK_LIBRARY}")
if(APPLE)
    set_target_properties(newyear-discord-social PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path")
elseif(UNIX)
    set_target_properties(newyear-discord-social PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
endif()
if(MSVC)
    target_compile_options(newyear-discord-social PRIVATE /utf-8)
endif()
add_custom_command(TARGET newyear-discord-social POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${DISCORD_SDK_RUNTIME}" "$<TARGET_FILE_DIR:newyear-discord-social>")
